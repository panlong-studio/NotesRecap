## 一、字符串处理

### 1. 字符串长度

#### strlen - 字符串长度

```c
#include <string.h>

size_t strlen(const char *s);
```

**参数解析：**

| 参数 | 类型         | 说明                           |
| :--- | :----------- | :----------------------------- |
| s    | const char * | 指向以空字符结尾的字符串的指针 |

**返回值解析：**

| 返回值     | 类型   | 说明                                     |
| :--------- | :----- | :--------------------------------------- |
| 字符串长度 | size_t | 返回字符串中字符个数（不包括结尾的'\0'） |

**示例：**

c

```
#include <stdio.h>
#include <string.h>

int main() {
    const char *str = "Hello World";
    printf("长度: %zu\n", strlen(str));  // 输出: 长度: 11
    return 0;
}
```

### 2. 字符串复制

#### strcpy/strncpy - 复制

c

```
#include <string.h>

char *strcpy(char *dest, const char *src);
char *strncpy(char *dest, const char *src, size_t n);
```



**参数解析：**

| 参数 | 类型         | 说明             |
| :--- | :----------- | :--------------- |
| dest | char *       | 目标缓冲区指针   |
| src  | const char * | 源字符串指针     |
| n    | size_t       | 最多复制的字节数 |

**返回值解析：**

| 返回值 | 类型   | 说明                 |
| :----- | :----- | :------------------- |
| dest   | char * | 返回目标缓冲区的指针 |

| 函数    | 说明                          |
| :------ | :---------------------------- |
| strcpy  | 复制整个src（含\0）           |
| strncpy | 最多复制n字节，可能不以\0结尾 |

**建议用法：**

c

```
strncpy(dest, src, sizeof(dest)-1);
dest[sizeof(dest)-1] = '\0';
```



**示例：**

c

```
#include <stdio.h>
#include <string.h>

int main() {
    char src[] = "Hello";
    char dest[20];

    strcpy(dest, src);
    printf("strcpy: %s\n", dest);  // Hello

    strncpy(dest, src, 3);
    dest[3] = '\0';
    printf("strncpy: %s\n", dest);  // Hel
    return 0;
}
```

### 3. 字符串连接

#### strcat/strncat - 连接

c

```
#include <string.h>

char *strcat(char *dest, const char *src);
char *strncat(char *dest, const char *src, size_t n);
```



**参数解析：**

| 参数 | 类型         | 说明                         |
| :--- | :----------- | :--------------------------- |
| dest | char *       | 目标字符串（必须有足够空间） |
| src  | const char * | 源字符串                     |
| n    | size_t       | 最多追加的字符数             |

**返回值解析：**

| 返回值 | 类型   | 说明                 |
| :----- | :----- | :------------------- |
| dest   | char * | 返回目标字符串的指针 |

**安全用法：**

c

```
n = sizeof(dest) - strlen(dest) - 1;
strncat(dest, src, n);
```



**示例：**

c

```
#include <stdio.h>
#include <string.h>

int main() {
    char dest[20] = "Hello";
    char src[] = " World";

    strcat(dest, src);
    printf("%s\n", dest);  // Hello World

    strncat(dest, "!!!", 2);
    printf("%s\n", dest);  // Hello World!!
    return 0;
}
```

### 4. 字符串比较

#### strcmp/strncmp - 比较

c

```
#include <string.h>

int strcmp(const char *s1, const char *s2);
int strncmp(const char *s1, const char *s2, size_t n);
```



**参数解析：**

| 参数 | 类型         | 说明             |
| :--- | :----------- | :--------------- |
| s1   | const char * | 第一个字符串     |
| s2   | const char * | 第二个字符串     |
| n    | size_t       | 最多比较的字符数 |

**返回值解析：**

| 返回值 | 类型 | 说明       |
| :----- | :--- | :--------- |
| 0      | int  | 字符串相等 |
| <0     | int  | s1 小于 s2 |
| >0     | int  | s1 大于 s2 |

**示例：**

c

```
#include <stdio.h>
#include <string.h>

int main() {
    printf("%d\n", strcmp("apple", "banana"));  // 负数
    printf("%d\n", strcmp("hello", "hello"));   // 0
    printf("%d\n", strncmp("abc123", "abc456", 3));  // 0
    return 0;
}
```

### 5. 字符分类（ctype.h）

c

```
#include <ctype.h>

int islower(int c);   // 小写字母
int isupper(int c);   // 大写字母
int isalpha(int c);   // 字母
int isdigit(int c);   // 数字
int isalnum(int c);   // 字母或数字
int isspace(int c);   // 空白字符
int toupper(int c);   // 转大写
int tolower(int c);   // 转小写
```



**参数解析：**

| 参数 | 类型 | 说明                                     |
| :--- | :--- | :--------------------------------------- |
| c    | int  | 要检查的字符（以unsigned char或EOF传递） |

**返回值解析：**

| 函数            | 返回值类型 | 说明                       |
| :-------------- | :--------- | :------------------------- |
| isxxx()         | int        | 满足条件返回非0，否则返回0 |
| toupper/tolower | int        | 返回转换后的字符           |

**示例：**

c

```
#include <stdio.h>
#include <ctype.h>

int main() {
    char c = 'a';
    printf("islower: %d\n", islower(c));  // 非0
    printf("toupper: %c\n", toupper(c));  // A

    char d = '5';
    printf("isdigit: %d\n", isdigit(d));  // 非0
    return 0;
}
```

## 二、文件操作

### 1. 无缓冲文件流

无缓冲文件流是直接与内核交互的文件操作方式，使用文件描述符。

#### open - 打开/创建文件

c

```
#include <fcntl.h>
#include <sys/types.h>
#include <sys/stat.h>

int open(const char *pathname, int flags);
int open(const char *pathname, int flags, mode_t mode);
```



**参数解析：**

| 参数     | 类型         | 说明                                             |
| :------- | :----------- | :----------------------------------------------- |
| pathname | const char * | 文件路径名                                       |
| flags    | int          | 打开标志（必须包含O_RDONLY/O_WRONLY/O_RDWR之一） |
| mode     | mode_t       | 文件权限（使用O_CREAT时必需）                    |

**flags常用标志：**

| 标志     | 说明                          |
| :------- | :---------------------------- |
| O_RDONLY | 只读                          |
| O_WRONLY | 只写                          |
| O_RDWR   | 可读可写                      |
| O_CREAT  | 文件不存在则创建              |
| O_EXCL   | 与O_CREAT配合，文件存在则失败 |
| O_TRUNC  | 截断文件为空                  |
| O_APPEND | 追加模式                      |

**返回值解析：**

| 返回值 | 类型 | 说明                       |
| :----- | :--- | :------------------------- |
| 成功   | int  | 返回文件描述符（非负整数） |
| 失败   | int  | 返回-1，并设置errno        |

**示例：**

c

```
#include <stdio.h>
#include <fcntl.h>
#include <unistd.h>

int main() {
    int fd = open("test.txt", O_CREAT | O_RDWR, 0644);
    if (fd == -1) {
        perror("open");
        return 1;
    }
    printf("文件描述符: %d\n", fd);
    close(fd);
    return 0;
}
```

#include <stdio.h>
#include <fcntl.h>
#include <unistd.h>

    return 0;
}

#### readdir - 读取目录项

c

```
#include <dirent.h>

struct dirent *readdir(DIR *dirp);
```



**参数解析：**

| 参数 | 类型  | 说明       |
| :--- | :---- | :--------- |
| dirp | DIR * | 目录流指针 |

**返回值解析：**

| 返回值   | 类型            | 说明                  |
| :------- | :-------------- | :-------------------- |
| 成功     | struct dirent * | 返回目录项结构体指针  |
| 目录读完 | struct dirent * | 返回NULL（errno不变） |
| 出错     | struct dirent * | 返回NULL，并设置errno |

**dirent结构体：**

c

```
struct dirent {
    ino_t          d_ino;       // inode编号
    off_t          d_off;       // 到下一个目录项的偏移
    unsigned short d_reclen;    // 目录项实际大小
    unsigned char  d_type;      // 文件类型
    char           d_name[256]; // 文件名
};
```



**d_type取值（文件类型）：**

| 值         | 类型     |
| :--------- | :------- |
| DT_BLK     | 块设备   |
| DT_CHR     | 字符设备 |
| DT_DIR     | 目录     |
| DT_FIFO    | 有名管道 |
| DT_LNK     | 符号链接 |
| DT_REG     | 普通文件 |
| DT_SOCK    | 套接字   |
| DT_UNKNOWN | 未知     |

**示例（遍历目录）：**

c

```
#include <stdio.h>
#include <dirent.h>

int main() {
    DIR *dir = opendir(".");
    if (dir == NULL) {
        perror("opendir");
        return 1;
    }

    struct dirent *entry;
    while ((entry = readdir(dir)) != NULL) {
        printf("%s\n", entry->d_name);
    }

    closedir(dir);
    return 0;
}
```

| 成功   | int  | 返回0               |

**stat结构体（主要成员）：**

c

```
struct stat {
    mode_t    st_mode;    // 文件类型和权限
    nlink_t   st_nlink;   // 硬链接数
    uid_t     st_uid;     // 所有者用户ID
    gid_t     st_gid;     // 所有者组ID
    off_t     st_size;    // 文件大小
    struct timespec st_mtim; // 最后修改时间
};
```



**示例：**

c

```
#include <stdio.h>
#include <sys/stat.h>

int main() {
    struct stat st;
    if (stat("test.txt", &st) == 0) {
        printf("文件大小: %ld bytes\n", st.st_size);
        printf("硬链接数: %lu\n", st.st_nlink);
    }
    return 0;
}
```

### 3. 文件映射

文件映射允许像访问内存一样访问文件内容。

#### mmap - 建立文件映射

c

```
#include <sys/mman.h>

void *mmap(void *addr, size_t length, int prot, int flags, int fd, off_t offset);
```



**参数解析：**

| 参数   | 类型   | 说明                               |
| :----- | :----- | :--------------------------------- |
| addr   | void * | 建议映射地址（通常NULL让系统选择） |
| length | size_t | 映射区域大小（字节）               |
| prot   | int    | 映射区域权限                       |
| flags  | int    | 映射特性                           |
| fd     | int    | 文件描述符                         |
| offset | off_t  | 文件内偏移量（必须是页大小的倍数） |

**prot取值：**

| 值                    | 说明     |
| :-------------------- | :------- |
| PROT_READ             | 可读     |
| PROT_WRITE            | 可写     |
| PROT_READ\|PROT_WRITE | 可读可写 |

**flags取值：**

| 值          | 说明                           |
| :---------- | :----------------------------- |
| MAP_SHARED  | 修改反映到文件，对其他进程可见 |
| MAP_PRIVATE | 修改不反映到文件，私有副本     |

**返回值解析：**

| 返回值 | 类型   | 说明                                     |
| :----- | :----- | :--------------------------------------- |
| 成功   | void * | 返回映射内存地址                         |
| 失败   | void * | 返回MAP_FAILED（(void*)-1），并设置errno |

**示例：**

c

```
#include <stdio.h>
#include <fcntl.h>
#include <sys/mman.h>
#include <string.h>
#include <unistd.h>

int main() {
    int fd = open("test.txt", O_RDWR | O_CREAT, 0644);
    if (fd == -1) return 1;
    
    // 设置文件大小
    ftruncate(fd, 4096);
    
    // 映射文件
    char *map = mmap(NULL, 4096, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
    if (map == MAP_FAILED) {
        perror("mmap");
        return 1;
    }
    
    // 像内存一样操作
    strcpy(map, "Hello from mmap!");
    
    // 解除映射
    munmap(map, 4096);
    close(fd);
    return 0;
}
```

void *producer(void *arg) {
    int item = 0;
    while (1) {
        pthread_mutex_lock(&mutex);
        while (count == BUFFER_SIZE) {
            pthread_cond_wait(&cond, &mutex);
        }
        buffer[count++] = item++;
        printf("生产: %d, 数量: %d\n", item-1, count);
        pthread_cond_signal(&cond);
        pthread_mutex_unlock(&mutex);
        sleep(1);
    }
    return NULL;

    while (1) {
        pthread_mutex_lock(&mutex);
        while (count == 0) {
            pthread_cond_wait(&cond, &mutex);
        }
        int item = buffer[--count];
        printf("消费: %d, 剩余: %d\n", item, count);
        pthread_cond_signal(&cond);
        pthread_mutex_unlock(&mutex);
        sleep(2);
    }
    return NULL;

    ftruncate(fd, 4096);
    
    char *map = mmap(NULL, 4096, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
    if (map == MAP_FAILED) return 1;
    
    strcpy(map, "POSIX共享内存");
    
    munmap(map, 4096);
    close(fd);
    // shm_unlink("/myshm");  // 删除
    return 0;
}

#include <stdio.h>
#include <string.h>
#include <unistd.h>
#include <arpa/inet.h>

int main() {
    int server_fd, client_fd;
    struct sockaddr_in server_addr, client_addr;
    socklen_t client_len = sizeof(client_addr);
    char buffer[1024] = {0};
    
    // 创建socket
    server_fd = socket(AF_INET, SOCK_STREAM, 0);
    
    // 设置地址复用
    int opt = 1;
    setsockopt(server_fd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));
    
    // 绑定
    server_addr.sin_family = AF_INET;
    server_addr.sin_addr.s_addr = INADDR_ANY;
    server_addr.sin_port = htons(8080);
    bind(server_fd, (struct sockaddr *)&server_addr, sizeof(server_addr));
    
    // 监听
    listen(server_fd, 5);
    printf("服务器启动，监听端口8080\n");
    
    // 接受连接
    client_fd = accept(server_fd, (struct sockaddr *)&client_addr, &client_len);
    printf("客户端连接: %s:%d\n", inet_ntoa(client_addr.sin_addr), ntohs(client_addr.sin_port));
    
    // 接收数据
    read(client_fd, buffer, sizeof(buffer));
    printf("收到: %s\n", buffer);
    
    // 发送响应
    char *response = "HTTP/1.1 200 OK\r\nContent-Length: 12\r\n\r\nHello World!";
    write(client_fd, response, strlen(response));
    
    close(client_fd);
    close(server_fd);
    return 0;
}

        FD_ZERO(&readfds);
        FD_SET(STDIN_FILENO, &readfds);

#### epoll_wait - 等待事件

c

```
#include <sys/epoll.h>

int epoll_wait(int epfd, struct epoll_event *events, int maxevents, int timeout);
```



**参数解析：**

| 参数      | 类型                 | 说明                                    |
| :-------- | :------------------- | :-------------------------------------- |
| epfd      | int                  | epoll文件描述符                         |
| events    | struct epoll_event * | 就绪事件数组                            |

**返回值解析：**

| :----- | :--- | :---------------- |
| >0     | int  | 就绪事件数量      |
| 0      | int  | 超时              |
| -1     | int  | 出错，并设置errno |

**示例（epoll服务器）：**

c

```
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <arpa/inet.h>
#include <sys/epoll.h>

#define MAX_EVENTS 10
#define PORT 8080

int main() {
    int server_fd, client_fd, epfd, nfds;
    struct sockaddr_in addr;
    struct epoll_event ev, events[MAX_EVENTS];
    
    // 创建socket
    server_fd = socket(AF_INET, SOCK_STREAM, 0);
    
    // 设置地址复用
    int opt = 1;
    setsockopt(server_fd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));
    
    // 绑定
    addr.sin_family = AF_INET;
    addr.sin_addr.s_addr = INADDR_ANY;
    addr.sin_port = htons(PORT);
    bind(server_fd, (struct sockaddr *)&addr, sizeof(addr));
    
    // 监听
    listen(server_fd, 5);
    
    // 创建epoll
    epfd = epoll_create(1);
    
    // 添加server_fd到epoll
    ev.events = EPOLLIN;
    ev.data.fd = server_fd;
    epoll_ctl(epfd, EPOLL_CTL_ADD, server_fd, &ev);
    
    printf("epoll服务器启动，端口%d\n", PORT);
    
    while (1) {
        nfds = epoll_wait(epfd, events, MAX_EVENTS, -1);
        
        for (int i = 0; i < nfds; i++) {
            if (events[i].data.fd == server_fd) {
                // 新连接
                client_fd = accept(server_fd, NULL, NULL);
                ev.events = EPOLLIN;
                ev.data.fd = client_fd;
                epoll_ctl(epfd, EPOLL_CTL_ADD, client_fd, &ev);
                printf("新客户端连接\n");
            } else {
                // 客户端数据
                char buf[1024];
                int n = read(events[i].data.fd, buf, sizeof(buf));
                if (n <= 0) {
                    close(events[i].data.fd);
                    epoll_ctl(epfd, EPOLL_CTL_DEL, events[i].data.fd, NULL);
                    printf("客户端断开\n");
                } else {
                    buf[n] = '\0';
                    printf("收到: %s", buf);
                    write(events[i].data.fd, buf, n);
                }
            }
        }
    }
    
    close(server_fd);
    return 0;
}
```

        return 1;

        fprintf(stderr, "连接失败: %s\n", mysql_error(conn));
        mysql_close(conn);
        return 1;
    }
    
    // 执行查询
    if (mysql_query(conn, "SELECT id, name FROM users")) {
        fprintf(stderr, "查询失败: %s\n", mysql_error(conn));
